        function openQuiz() {

            modal.classList.remove('invisible', 'opacity-0');

            setTimeout(() => { modalInner.classList.remove('scale-95', 'opacity-0'); }, 50);

            widget.classList.add('translate-y-24', 'opacity-0');

        }


        function closeQuiz() {

            modalInner.classList.add('scale-95', 'opacity-0');

            setTimeout(() => {

                modal.classList.add('invisible', 'opacity-0');

                if(currentQuestionId !== "result") { widget.classList.remove('translate-y-24', 'opacity-0'); }

            }, 400);

        }


        function startQuiz() {

            introView.classList.add('hidden');

            questionView.classList.remove('hidden');

            currentQuestionId = "q1";

            questionStep = 1;

            for(let key in scores) scores[key] = 0;

            renderQuestion();

        }


        function renderQuestion() {

            const progressPercentage = ((questionStep - 1) / totalSteps) * 100;

            progressBar.style.width = `${progressPercentage}%`;

            if (currentQuestionId === "result") { showResults(); return; }

            const q = quizData[currentQuestionId];

            document.getElementById('q-counter').innerText = `Question ${questionStep} of ${totalSteps}`;

            document.getElementById('q-text').innerText = q.question;

            const optionsContainer = document.getElementById('q-options');

            optionsContainer.innerHTML = '';

            q.options.forEach((opt, index) => {

                const btn = document.createElement('button');

                btn.className = "w-full text-left p-4 rounded-xl border-2 border-gray-100 bg-white text-brand-charcoal text-sm sm:text-base font-medium transition-all duration-200 quiz-option-hover relative flex items-center";

                btn.innerHTML = `<span class="w-6 h-6 rounded-full bg-gray-100 text-gray-500 flex items-center justify-center text-xs font-bold mr-3 shrink-0">${String.fromCharCode(65 + index)}</span> <span>${opt.text}</span>`;

                btn.onclick = () => selectOption(opt.type, opt.next, btn);

                optionsContainer.appendChild(btn);

            });

            optionsContainer.style.opacity = 0;

            setTimeout(() => { optionsContainer.style.transition = 'opacity 0.3s'; optionsContainer.style.opacity = 1; }, 50);

        }


        function selectOption(type, nextQuestionId, btnElement) {

            const allBtns = document.getElementById('q-options').children;

            for(let b of allBtns) { b.classList.remove('quiz-option-selected'); b.classList.add('opacity-50'); }

            btnElement.classList.add('quiz-option-selected');

            btnElement.classList.remove('opacity-50');

            if(scores[type] !== undefined) { scores[type]++; }

            setTimeout(() => {

                currentQuestionId = nextQuestionId;

                if(nextQuestionId !== "result") { questionStep++; }

                renderQuestion();

            }, 400);

        }


        function showResults() {

            questionView.classList.add('hidden');

            resultView.classList.remove('hidden');

            progressBar.style.width = '100%';

            let maxScore = 0;

            let dominantTrait = 'confusion';

            const painPoints = ['confusion', 'resume', 'interview', 'direction', 'strategy'];

            painPoints.forEach(trait => { if(scores[trait] > maxScore) { maxScore = scores[trait]; dominantTrait = trait; } });

            const resultData = getResultContent(dominantTrait);

            document.getElementById('result-icon').innerText = resultData.emoji;

            document.getElementById('result-title').innerText = resultData.title;

            document.getElementById('result-message').innerText = resultData.message;

        }
